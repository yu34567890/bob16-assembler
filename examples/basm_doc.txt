-- built in instructions --
(these instruction exist in the cpu)
(and these instructions are different from bob16 docs)

BR: if condition codes match any of the nzp flags, pc += imm[9]
	ex. br 0b101 50
		(jumps to incremented pc + 50 if condition codes n or p are 1)
	flag codes
	0b001 p
	0b010 z
	0b100 n 
	nzp does not works in br only binary values 

LEA: dst = pc + imm
	ex. LEA R7 10 
		(sets register 7 to pc + 10)
	

JMPR: pc = r[src]
	ex. jmpr r0
	(only register jumping)

JSR: r7 = pc and pc = register / label / 11 bit imm
	ex. 
	LOOP:
	jsr LOOP:

	can be used for function calls too


-- psudeo instructions --
(these instructions does not exist on the cpu but assembler transpiles them to valid bob16 instructions)

LI: loads an imm to dst register
	ex.
	li r1 0xffff
	r1 now contains 0xffff
	labels can be used too 
	li r1 SOME_FUNC
	r1 now contains adress of that label 

LP: load an memory adress to dst register
	ex.
	lp r1 0xffff
	r1 now contains value of memory adress 0xffff (does not support registers )
	labels can be used here too

WRT: write dst to memory adress at r1
	ex. 
	write r1 r2
	memory[r2] = r1

-- stdlib instructions --
(these instructions are defined in the stdlib)

%include "stdlib.basm"

(important note stdlib uses r0 as stack pointer main.basm or any other program shall not use r0 ever same as r7 as r7 is return to caller)
(stack starts at 0x8000 tho can be changed by changing the value of r0 you should put li r0 stack_adress so stdlib will use that adress )

push r1 ; pushes r1 to memory adress at r0 and increments r0

pop r1 : decrements r0 and reads it to r1 

cmp: compares given registers and sets the flags
	ex. 
	cmp r1 r2 
	if r1 > r2
		p = 1 
	if r1 < r2
		n = 1 
	if r1 = r2
		z = 1 
	
	-- WARN --
	you should use cmp_reverse after using cmp otherwise 2 registers will get corrupted
	ex 
	cmp r1 r2
	br 0b001 label ; if r1 > r2 jmp to label 


	label:
	cmp_reverse r1 r2


	and you should use br right after using the cmp instruction otherwise the flags will be reset by an another instruction 

	cmp r1 r2 
	add r3 r1 r2 
	br 0b001 label
	; this is wrong as this uses the broken r1 r2 values and br will not jmp to the label
